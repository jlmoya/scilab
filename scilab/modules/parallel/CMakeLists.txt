# CMake build for the `parallel` module — Stage-1 make→CMake roll-out #2, the
# external-dependency exemplar (OpenMP/libomp via find_package).
#
# HYBRID COEXISTENCE: the rest of the tree stays autotools; this standalone
# project builds ONLY libsciparallel and its dylib is dropped into
# modules/parallel/.libs/ in place of the autotools one (`parallel` is
# dlopen-only — no executable links it — so nothing else needs rewiring).
# This file is invisible to automake; the autotools parallel is always
# recoverable with `make -C modules/parallel`.
#
# PARITY CONTRACT (arbitrated by build-parity/): the output must be
# behaviorally identical to the autotools build —
#   file:          libsciparallel.2027.dylib (+ libsciparallel.dylib symlink)
#   install_name:  /usr/local/lib/scilab/libsciparallel.2027.dylib
#   exports:       _gw_parallel _parallelConcurrency _parallelForbidden (exactly these 3)
#   deps:          /usr/lib/libSystem.B.dylib
#                  + /opt/homebrew/opt/libomp/lib/libomp.dylib  (THE new dimension)
# Every compiler/linker fact below is transcribed from the CONFIGURED autotools
# build (config.status SCI_CFLAGS + modules/parallel/Makefile), not invented here.
#
# Build:    cmake -S . -B build-cmake && cmake --build build-cmake
# Drop-in:  cmake --build build-cmake --target drop-in
# Verify:   cd ../../build-parity && ./capture.sh .. /tmp/cand.json cand \
#           && ./diff.sh baseline-autotools.json /tmp/cand.json

cmake_minimum_required(VERSION 3.20)

# Autotools compiles with CC="gcc -std=gnu23 -arch arm64" — on this platform
# /usr/bin/gcc IS Apple clang. Default to the same driver (honor an explicit
# -DCMAKE_C_COMPILER= or $CC override).
if(NOT DEFINED CMAKE_C_COMPILER AND NOT DEFINED ENV{CC})
    set(CMAKE_C_COMPILER gcc)
endif()
set(CMAKE_OSX_ARCHITECTURES arm64)        # CC's -arch arm64
set(CMAKE_OSX_DEPLOYMENT_TARGET 11.0)     # -mmacosx-version-min=11.0 (compile + link)

project(scilab-parallel C)

# The parity harness reads the compile command from compile_commands.json to
# verify the semantic flag facts (opt/wrapv/min_macos/openmp).
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Root of the Scilab source tree (this file lives in modules/parallel/).
# Overridable so an out-of-tree checkout can still point at a configured tree
# (machine.h/version.h below are generated by configure, not by us).
set(SCILAB_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH
    "Scilab source tree root (must be autotools-configured: modules/core/includes/machine.h)")
get_filename_component(SCILAB_SOURCE_DIR "${SCILAB_SOURCE_DIR}" ABSOLUTE)
if(NOT EXISTS "${SCILAB_SOURCE_DIR}/modules/core/includes/machine.h")
    message(FATAL_ERROR "SCILAB_SOURCE_DIR (${SCILAB_SOURCE_DIR}) is not a configured "
                        "Scilab tree: modules/core/includes/machine.h not found. "
                        "Run ./configure there first (machine.h is configure-generated).")
endif()

# THE NEW DIMENSION vs the sound beachhead: an external dependency, resolved by
# CMake's own machinery instead of configure's hardcoded flags. Autotools
# compiled with OPENMP_CXXFLAGS="-Xpreprocessor -fopenmp" and linked
# OPENMP_LIBS="-lomp" against -L/opt/homebrew/opt/libomp/lib (global LDFLAGS).
# Here FindOpenMP produces the equivalent: AppleClang gets the "-Xclang
# -fopenmp" spelling (same token-wise `-fopenmp` fact the parity harness keys
# on; noparallel.c has no omp pragmas, so frontend-vs-preprocessor makes zero
# codegen difference) and the omp library resolved to the Homebrew keg. Apple
# clang ships no libomp, so the keg is the default root (an explicit
# -DOpenMP_ROOT= or env override still wins, CMP0074).
#
# Parity note: linking libomp by absolute path records the dep at libomp's
# install_name — /opt/homebrew/opt/libomp/lib/libomp.dylib — exactly what the
# autotools `-lomp` link recorded (NOT an @rpath form). The harness verifies it.
if(NOT DEFINED OpenMP_ROOT AND NOT DEFINED ENV{OpenMP_ROOT})
    set(OpenMP_ROOT "/opt/homebrew/opt/libomp")
endif()
find_package(OpenMP REQUIRED COMPONENTS C)
get_target_property(_openmp_link_libs OpenMP::OpenMP_C INTERFACE_LINK_LIBRARIES)
message(STATUS "OpenMP C flags: ${OpenMP_C_FLAGS}; link libraries: ${_openmp_link_libs}")

# libtool: -version-number $(SCILAB_LIBRARY_VERSION) with 2027:0:0 names the
# file libsciparallel.2027.dylib and stamps Mach-O compat/current 2028.0.0
# (darwin libtool uses major+minor+1). The 4-digit token in the FILENAME is
# what the parity harness keys on (libsciparallel.VER.dylib) — it must be
# exactly ".2027." with nothing after it before ".dylib".
set(SCILAB_LIBRARY_VERSION_MAJOR 2027)
math(EXPR _macho_version "${SCILAB_LIBRARY_VERSION_MAJOR} + 1")

# One C source. The Makefile.am calls it GATEWAY_CPP_SOURCES but it is a .c
# file (a C compile). src/c/DllmainParallel.c and src/c/forkWindows.c are
# Windows-only (the autotools build excludes them too); the real OpenMP
# gateways (sci_gateway/cpp/) are commented out upstream — the shipped
# parallel is the noparallel stub, on every platform.
add_library(sciparallel SHARED
    src/noparallel/noparallel.c)

# DEFS from the autotools compile line.
target_compile_definitions(sciparallel PRIVATE HAVE_CONFIG_H)

# Include order mirrors the autotools compile line
# $(DEFAULT_INCLUDES) $(libsciparallel_la_CPPFLAGS) $(CPPFLAGS), with ONE
# deliberate deviation: libomp's include is NOT listed here — OpenMP::OpenMP_C
# contributes it as a trailing -isystem (see the CPPFLAGS block below). CFLAGS'
# trailing -I/opt/homebrew/opt/openssl/include rides last in the compile
# options below (CMake emits includes before flags, matching that placement).
target_include_directories(sciparallel PRIVATE
    # DEFAULT_INCLUDES: -I. -I$(top_builddir)/modules/core/includes
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${SCILAB_SOURCE_DIR}/modules/core/includes      # configure-GENERATED machine.h/version.h
    # libsciparallel_la_CPPFLAGS (its lead item, $(OPENMP_CXXFLAGS), is carried
    # by OpenMP::OpenMP_C as a compile option — flag position is semantically
    # free; the Makefile also repeats modules/core/includes here, which CMake
    # de-duplicates against DEFAULT_INCLUDES above — same preprocessor result):
    ${CMAKE_CURRENT_SOURCE_DIR}/includes
    ${CMAKE_CURRENT_SOURCE_DIR}/src/cpp
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/ast
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/exps
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/operations
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/parse
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/symbol
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/system_env
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/types
    ${SCILAB_SOURCE_DIR}/modules/api_scilab/includes
    ${SCILAB_SOURCE_DIR}/modules/localization/includes
    ${SCILAB_SOURCE_DIR}/modules/output_stream/includes
    ${SCILAB_SOURCE_DIR}/modules/parameters/includes
    ${SCILAB_SOURCE_DIR}/modules/dynamic_link/includes
    # CPPFLAGS (configure-detected Homebrew paths). libomp's include
    # (/opt/homebrew/opt/libomp/include, in autotools' CPPFLAGS) is deliberately
    # omitted: OpenMP::OpenMP_C already contributes it as a trailing -isystem, so
    # a hardcoded -I here is dead (CMake de-duplicates it) AND a copier trap —
    # under an OpenMP_ROOT override the hardcoded keg -I would silently outrank
    # the override for a module that #includes <omp.h>. Let the imported target
    # own it.
    /opt/homebrew/include
    /opt/homebrew/opt/libarchive/include)

# CC's -std=gnu23, then SCI_CFLAGS VERBATIM from config.status (do NOT replace
# with CMAKE_BUILD_TYPE presets — parity is on these exact semantic facts:
# O2 + fwrapv + min_macos 11.0), then configure's CFLAGS (an include path).
target_compile_options(sciparallel PRIVATE
    -std=gnu23
    -DNDEBUG -g1 -O2 -fwrapv -mmacosx-version-min=11.0 -fno-stack-protector
    -Wall -Wpedantic -Werror=implicit -Werror=incompatible-pointer-types
    -I/opt/homebrew/opt/openssl/include)

# OpenMP::OpenMP_C contributes the compile flag ("-Xclang -fopenmp" — the
# harness's openmp=True fact) and the libomp link entry (autotools: -lomp).
# libtool's darwin allow_undefined_flag, VERBATIM: the gateway calls Scilab
# API symbols (Scierror, ...) that live in the hosting process — they resolve
# at dlopen time, never at link time. This is precisely why the autotools
# dylib's deps are libSystem + libomp only.
# -no_fixup_chains rides along (dynamic_lookup is incompatible with chained
# fixups; libtool disables them, and it changes the dyld load commands).
target_link_libraries(sciparallel PRIVATE OpenMP::OpenMP_C)
target_link_options(sciparallel PRIVATE
    "LINKER:-undefined,dynamic_lookup"
    "LINKER:-no_fixup_chains"
    # Global configured LDFLAGS rpaths — libtool stamped these two LC_RPATHs
    # into the dylib. (Its -L search dirs are dropped: libomp is linked by
    # absolute path above, and with no other -l libraries they add no load
    # command. Global LIBS=-ldl likewise folds into libSystem on macOS. A C++
    # link would drag in libc++ — this project is C-only, so the C driver
    # links.)
    "LINKER:-rpath,/usr/lib"
    "LINKER:-rpath,/opt/homebrew/opt/gcc/lib/gcc/current")

set_target_properties(sciparallel PROPERTIES
    # EXACT filename libsciparallel.2027.dylib: version baked into OUTPUT_NAME.
    # Deliberately NOT CMake VERSION/SOVERSION — those would name it
    # libsciparallel.2027.0.0.dylib, which breaks the harness key.
    OUTPUT_NAME "sciparallel.${SCILAB_LIBRARY_VERSION_MAJOR}"
    # Drop CMake's automatic -Dsciparallel_EXPORTS: autotools defines no such
    # symbol (it would only matter under _MSC_VER anyway — dynlib_parallel.h).
    DEFINE_SYMBOL ""
    SUFFIX ".dylib"
    # libtool's `-rpath $(pkglibdir)` == absolute install_name, not @rpath.
    INSTALL_NAME_DIR "/usr/local/lib/scilab"
    BUILD_WITH_INSTALL_NAME_DIR TRUE
    # libtool 2027:0:0 → compat/current 2028.0.0 (harness strips these, but
    # match them anyway for bit-level load-command fidelity).
    MACHO_COMPATIBILITY_VERSION "${_macho_version}.0.0"
    MACHO_CURRENT_VERSION "${_macho_version}.0.0")

# libtool also leaves an unversioned symlink next to the versioned dylib.
add_custom_command(TARGET sciparallel POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E create_symlink
            $<TARGET_FILE_NAME:sciparallel> ${CMAKE_CURRENT_BINARY_DIR}/libsciparallel.dylib
    COMMENT "Symlinking libsciparallel.dylib -> $<TARGET_FILE_NAME:sciparallel>")

# The hybrid drop-in: put the CMake-built dylib where the runtime dlopens it
# (modules/parallel/.libs/ — hardcoded fallback path) and where the parity
# harness fingerprints it. A REAL copy, not a symlink: the harness skips
# symlinked dylibs, and the autotools original must remain recoverable by
# simply rebuilding (`make -C modules/parallel`).
add_custom_target(drop-in
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_SOURCE_DIR}/.libs
    COMMAND ${CMAKE_COMMAND} -E copy
            $<TARGET_FILE:sciparallel> ${CMAKE_CURRENT_SOURCE_DIR}/.libs/$<TARGET_FILE_NAME:sciparallel>
    COMMAND ${CMAKE_COMMAND} -E create_symlink
            $<TARGET_FILE_NAME:sciparallel> ${CMAKE_CURRENT_SOURCE_DIR}/.libs/libsciparallel.dylib
    DEPENDS sciparallel
    COMMENT "Dropping CMake-built parallel into modules/parallel/.libs/ (hybrid coexistence)")
