# CMake build for the `sound` module — the Stage-1 make→CMake beachhead.
#
# HYBRID COEXISTENCE: the rest of the tree stays autotools; this standalone
# project builds ONLY libscisound and its dylib is dropped into
# modules/sound/.libs/ in place of the autotools one (`sound` is dlopen-only —
# no executable links it — so nothing else needs rewiring). This file is
# invisible to automake; the autotools sound is always recoverable with
# `make -C modules/sound`.
#
# PARITY CONTRACT (arbitrated by build-parity/): the output must be
# behaviorally identical to the autotools build —
#   file:          libscisound.2027.dylib (+ libscisound.dylib symlink)
#   install_name:  /usr/local/lib/scilab/libscisound.2027.dylib
#   exports:       _doBeep _sci_PlaySound _sci_beep   (exactly these 3)
#   deps:          /usr/lib/libSystem.B.dylib only
# Every compiler/linker fact below is transcribed from the CONFIGURED autotools
# build (config.status SCI_CFLAGS + modules/sound/Makefile), not invented here.
#
# Build:    cmake -S . -B build-cmake && cmake --build build-cmake
# Drop-in:  cmake --build build-cmake --target drop-in
# Verify:   cd ../../build-parity && ./capture.sh .. /tmp/cand.json cand \
#           && ./diff.sh baseline-autotools.json /tmp/cand.json

cmake_minimum_required(VERSION 3.20)

# Autotools compiles with CC="gcc -std=gnu23 -arch arm64" — on this platform
# /usr/bin/gcc IS Apple clang. Default to the same driver (honor an explicit
# -DCMAKE_C_COMPILER= or $CC override).
if(NOT DEFINED CMAKE_C_COMPILER AND NOT DEFINED ENV{CC})
    set(CMAKE_C_COMPILER gcc)
endif()
set(CMAKE_OSX_ARCHITECTURES arm64)        # CC's -arch arm64
set(CMAKE_OSX_DEPLOYMENT_TARGET 11.0)     # -mmacosx-version-min=11.0 (compile + link)

project(scilab-sound C)

# The parity harness reads the compile command from compile_commands.json to
# verify the semantic flag facts (opt/wrapv/min_macos).
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Root of the Scilab source tree (this file lives in modules/sound/).
# Overridable so an out-of-tree checkout can still point at a configured tree
# (machine.h/version.h below are generated by configure, not by us).
set(SCILAB_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH
    "Scilab source tree root (must be autotools-configured: modules/core/includes/machine.h)")
get_filename_component(SCILAB_SOURCE_DIR "${SCILAB_SOURCE_DIR}" ABSOLUTE)
if(NOT EXISTS "${SCILAB_SOURCE_DIR}/modules/core/includes/machine.h")
    message(FATAL_ERROR "SCILAB_SOURCE_DIR (${SCILAB_SOURCE_DIR}) is not a configured "
                        "Scilab tree: modules/core/includes/machine.h not found. "
                        "Run ./configure there first (machine.h is configure-generated).")
endif()

# libtool: -version-number $(SCILAB_LIBRARY_VERSION) with 2027:0:0 names the
# file libscisound.2027.dylib and stamps Mach-O compat/current 2028.0.0
# (darwin libtool uses major+minor+1). The 4-digit token in the FILENAME is
# what the parity harness keys on (libscisound.VER.dylib) — it must be exactly
# ".2027." with nothing after it before ".dylib".
set(SCILAB_LIBRARY_VERSION_MAJOR 2027)
math(EXPR _macho_version "${SCILAB_LIBRARY_VERSION_MAJOR} + 1")

# Only the two gateway sources. src/c/DllmainSound.c and src/nosound/ are
# Windows-only (the autotools build excludes them too).
add_library(scisound SHARED
    sci_gateway/c/sci_beep.c
    sci_gateway/c/sci_PlaySound.c)

# DEFS from the autotools compile line.
target_compile_definitions(scisound PRIVATE HAVE_CONFIG_H)

# Include order mirrors the autotools compile line exactly:
# $(DEFAULT_INCLUDES) $(libscisound_la_CPPFLAGS) $(CPPFLAGS) — and CFLAGS'
# trailing -I/opt/homebrew/opt/openssl/include rides last in the compile
# options below (CMake emits includes before flags, matching that placement).
target_include_directories(scisound PRIVATE
    # DEFAULT_INCLUDES: -I. -I$(top_builddir)/modules/core/includes
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${SCILAB_SOURCE_DIR}/modules/core/includes      # configure-GENERATED machine.h/version.h
    # libscisound_la_CPPFLAGS:
    ${CMAKE_CURRENT_SOURCE_DIR}/includes
    ${CMAKE_CURRENT_SOURCE_DIR}/src/c
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/ast
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/exps
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/operations
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/parse
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/symbol
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/system_env
    ${SCILAB_SOURCE_DIR}/modules/ast/includes/types
    ${SCILAB_SOURCE_DIR}/modules/output_stream/includes
    ${SCILAB_SOURCE_DIR}/modules/fileio/includes
    ${SCILAB_SOURCE_DIR}/modules/api_scilab/includes
    ${SCILAB_SOURCE_DIR}/modules/localization/includes
    ${SCILAB_SOURCE_DIR}/modules/string/includes
    # CPPFLAGS (configure-detected Homebrew paths):
    /opt/homebrew/include
    /opt/homebrew/opt/libomp/include
    /opt/homebrew/opt/libarchive/include)

# CC's -std=gnu23, then SCI_CFLAGS VERBATIM from config.status (do NOT replace
# with CMAKE_BUILD_TYPE presets — parity is on these exact semantic facts:
# O2 + fwrapv + min_macos 11.0), then configure's CFLAGS (an include path).
target_compile_options(scisound PRIVATE
    -std=gnu23
    -DNDEBUG -g1 -O2 -fwrapv -mmacosx-version-min=11.0 -fno-stack-protector
    -Wall -Wpedantic -Werror=implicit -Werror=incompatible-pointer-types
    -I/opt/homebrew/opt/openssl/include)

# libtool's darwin allow_undefined_flag, VERBATIM: the gateway calls Scilab
# API symbols (Scierror, getVarAddressFromPosition, ...) that live in the
# hosting process — they resolve at dlopen time, never at link time. This is
# precisely why the autotools dylib's deps are libSystem-only.
# -no_fixup_chains rides along (dynamic_lookup is incompatible with chained
# fixups; libtool disables them, and it changes the dyld load commands).
target_link_options(scisound PRIVATE
    "LINKER:-undefined,dynamic_lookup"
    "LINKER:-no_fixup_chains"
    # SCI_LDFLAGS rpaths — libtool stamped these two LC_RPATHs into the dylib.
    # (Configure's -L search dirs are dropped: with no -l libraries they add
    # no load command. Global LIBS=-ldl likewise folds into libSystem on
    # macOS. Deps must stay libSystem-only; a C++ link would drag in libc++ —
    # this project is C-only, so the C driver links.)
    "LINKER:-rpath,/usr/lib"
    "LINKER:-rpath,/opt/homebrew/opt/gcc/lib/gcc/current")

set_target_properties(scisound PROPERTIES
    # EXACT filename libscisound.2027.dylib: version baked into OUTPUT_NAME.
    # Deliberately NOT CMake VERSION/SOVERSION — those would name it
    # libscisound.2027.0.0.dylib, which breaks the harness key.
    OUTPUT_NAME "scisound.${SCILAB_LIBRARY_VERSION_MAJOR}"
    # Drop CMake's automatic -Dscisound_EXPORTS: autotools defines no such
    # symbol (it would only matter under _MSC_VER anyway — dynlib_sound.h).
    DEFINE_SYMBOL ""
    SUFFIX ".dylib"
    # libtool's `-rpath $(pkglibdir)` == absolute install_name, not @rpath.
    INSTALL_NAME_DIR "/usr/local/lib/scilab"
    BUILD_WITH_INSTALL_NAME_DIR TRUE
    # libtool 2027:0:0 → compat/current 2028.0.0 (harness strips these, but
    # match them anyway for bit-level load-command fidelity).
    MACHO_COMPATIBILITY_VERSION "${_macho_version}.0.0"
    MACHO_CURRENT_VERSION "${_macho_version}.0.0")

# libtool also leaves an unversioned symlink next to the versioned dylib.
add_custom_command(TARGET scisound POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E create_symlink
            $<TARGET_FILE_NAME:scisound> ${CMAKE_CURRENT_BINARY_DIR}/libscisound.dylib
    COMMENT "Symlinking libscisound.dylib -> $<TARGET_FILE_NAME:scisound>")

# The hybrid drop-in: put the CMake-built dylib where the runtime dlopens it
# (modules/sound/.libs/ — hardcoded fallback path) and where the parity
# harness fingerprints it. A REAL copy, not a symlink: the harness skips
# symlinked dylibs, and the autotools original must remain recoverable by
# simply rebuilding (`make -C modules/sound`).
add_custom_target(drop-in
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_SOURCE_DIR}/.libs
    COMMAND ${CMAKE_COMMAND} -E copy
            $<TARGET_FILE:scisound> ${CMAKE_CURRENT_SOURCE_DIR}/.libs/$<TARGET_FILE_NAME:scisound>
    COMMAND ${CMAKE_COMMAND} -E create_symlink
            $<TARGET_FILE_NAME:scisound> ${CMAKE_CURRENT_SOURCE_DIR}/.libs/libscisound.dylib
    DEPENDS scisound
    COMMENT "Dropping CMake-built sound into modules/sound/.libs/ (hybrid coexistence)")
